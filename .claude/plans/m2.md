**Code:**
```js
// Which Claude profile an agent runs under, and on which machine. A SessionStart hook labels the terminal
// with the profile's key; each machine's ~/.config/latveria/profiles.json gives the key a label and color.
export const LABEL = "latveria.profile";
export const UNKNOWN = Object.freeze({ key: "?", label: "?", color: null });
const KEY = /^[A-Za-z0-9_-]{1,24}$/;
const COLOR = /^#[0-9a-fA-F]{6}$/;
const MAX = 32;
const isKey = (k) => typeof k === "string" && KEY.test(k);

// "laptop/t2" → "laptop"; a terminal here has no machine in its address.
export const machineOf = (address) => (typeof address === "string" && address.includes("/") ? address.split("/")[0] : null);

// The map is a local file the person wrote: anything broken gives no profiles, never an error.
export function parseProfiles(text) {
  let data;
  try { data = JSON.parse(text); } catch { return []; }
  const list = Array.isArray(data?.profiles) ? data.profiles : [];
  return list.filter((p) => isKey(p?.key)).slice(0, MAX).map((p) => ({
    key: p.key,
    label: typeof p.label === "string" && p.label.trim() ? p.label.trim().slice(0, 16) : p.key,
    color: typeof p.color === "string" && COLOR.test(p.color) ? p.color : null,
  }));
}

export function profileOf(terminal, profiles) {
  const key = terminal?.labels?.[LABEL];
  if (!isKey(key)) return UNKNOWN;
  return profiles.find((p) => p.key === key) ?? { key, label: key, color: null };
}
```

## T3: hook `latveria-label.sh` (decyzja właściciela 16.11: powłoka + `jq`, zgodnie z designem)

**Files:** `hooks/latveria-label.sh`, `test/hook.test.mjs`, `test/rules.test.mjs` (skaner obejmuje też `*.sh`)
**Code:**
```sh
#!/bin/sh
# Claude Code SessionStart hook: labels this Herdr terminal latveria.profile=<KEY>, the profile in
# ~/.config/latveria/profiles.json whose claudeDir is this session's ${CLAUDE_CONFIG_DIR:-~/.claude}.
# No network, no credentials. Outside Herdr, without jq, a map or a match it does nothing. It prints nothing
# (SessionStart output reaches Claude's context) and always exits 0, so it never stops Claude from starting.
[ -n "$HC_TERMINAL_ID" ] || exit 0
case "$HC_TERMINAL_ID" in -*|*[!A-Za-z0-9_./-]*) exit 0 ;; esac # never a flag for hc
command -v jq >/dev/null 2>&1 || exit 0
command -v hc >/dev/null 2>&1 || exit 0
map="${LATVERIA_STATE:-$HOME/.config/latveria}/profiles.json"
[ -r "$map" ] || exit 0

# A folder's real path: ~ expanded, symlinks and a trailing slash resolved; a missing folder keeps its spelling.
real() {
  case "$1" in "~") set -- "$HOME" ;; "~/"*) set -- "$HOME/${1#"~/"}" ;; esac
  (cd "$1" 2>/dev/null && pwd -P) || printf '%s\n' "${1%/}"
}
want=$(real "${CLAUDE_CONFIG_DIR:-$HOME/.claude}")
tab=$(printf '\t')

jq -r '(.profiles? // []) | if type == "array" then .[] else empty end
  | select(type == "object" and (.key | type) == "string" and (.claudeDir | type) == "string")
  | [.key, .claudeDir] | @tsv' "$map" 2>/dev/null |
while IFS="$tab" read -r key dir; do
  case "$key" in ""|*[!A-Za-z0-9_-]*) continue ;; esac
  [ "${#key}" -le 24 ] || continue
  [ "$(real "$dir")" = "$want" ] || continue
  hc label "$HC_TERMINAL_ID" "latveria.profile=$key" >/dev/null 2>&1
  break
done
exit 0
```

`PATH` = katalog z atrapą `hc` (`#!/bin/sh` + `echo "$@" >> "$HC_LOG"`, chmod 755) + katalog z `jq` (`dirname` z
