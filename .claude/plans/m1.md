```js
  const args = [...(argsSource ?? "").matchAll(/["'`]([^"'`]*)["'`]/g)].map((m) => m[1]);
```

```js
// A recording stand-in for GPUI elements: every builder call is kept, children are collected.
export function element(kind, fields = {}) {
  const node = { kind, ...fields, calls: [], kids: [] }; // not "children": that is a builder method
  const self = new Proxy(node, {
    get(target, prop) {
      if (prop in target || typeof prop === "symbol" || prop === "then") return target[prop];
      return (...args) => {
        if (prop === "child") target.kids.push(args[0]);
        else if (prop === "children") target.kids.push(...args[0]);
        else if (prop === "when") { if (args[0]) args[1](self); }
        else target.calls.push([prop, ...args]);
        return self;
      };
    },
  });
  return self;
}

// Every element under `root`, depth first.
export function nodes(root) {
  const out = [];
  const walk = (n) => { if (n && typeof n === "object" && "kind" in n) { out.push(n); n.kids.forEach(walk); } };
  walk(root);
  return out;
}

// The text an element shows, its strings joined by spaces.
export function text(root) {
  const out = [];
  const walk = (n) => {
    if (typeof n === "string" || typeof n === "number") out.push(String(n));
    else if (n?.kids) n.kids.forEach(walk);
  };
  walk(root);
  return out.join(" ");
}
```

```js
#!/usr/bin/env node
// After a Herdr update: has a mod we cloned changed upstream since we cloned it? Compares
// upstream/<name>-<version>/main.js (the pristine copy, from mod.json's cloned_from) with the
// installed Herdr mod. Exit 0 when nothing changed, 1 when something did or is missing.
import { execFileSync } from "node:child_process";
import { existsSync, readFileSync, readdirSync } from "node:fs";
import { homedir } from "node:os";
import { join } from "node:path";
import { fileURLToPath } from "node:url";

function gitDiff(a, b) {
  try {
    execFileSync("git", ["diff", "--no-index", "--no-color", "--", a, b], { encoding: "utf8" });
    return "";
  } catch (e) {
    return e.stdout ?? String(e); // git diff exits 1 when the files differ
  }
}

export function upstreamChanges({ repo, herdrHome }) {
  const mods = join(repo, "mods");
  if (!existsSync(mods)) return [];
  const out = [];
  for (const mod of readdirSync(mods).sort()) {
    const metaPath = join(mods, mod, "mod.json");
    if (!existsSync(metaPath)) continue;
    const from = JSON.parse(readFileSync(metaPath, "utf8")).cloned_from;
    if (!from) continue;
    const pristine = join(repo, "upstream", `${from.name}-${from.version}`, "main.js");
    const current = join(herdrHome, "mods", from.name, "main.js");
    if (!existsSync(pristine) || !existsSync(current)) {
      out.push({ mod, from, status: "missing", diff: "" });
      continue;
    }
    const same = readFileSync(pristine, "utf8") === readFileSync(current, "utf8");
    out.push({ mod, from, status: same ? "same" : "changed", diff: same ? "" : gitDiff(pristine, current) });
  }
  return out;
}

if (process.argv[1] === fileURLToPath(import.meta.url)) {
  const results = upstreamChanges({
    repo: fileURLToPath(new URL("..", import.meta.url)),
    herdrHome: process.env.HERDR_APP_HOME ?? join(homedir(), ".config", "herdr-app"),
  });
  for (const r of results) {
    console.log(`${r.status.padEnd(7)} ${r.mod} (cloned from ${r.from.name} ${r.from.version})`);
    if (r.diff) console.log(r.diff);
  }
  process.exit(results.some((r) => r.status !== "same") ? 1 : 0);
}
```

```js
#!/usr/bin/env node
// Vendors Material Icon Theme (MIT) into the Files mod: the dark icons its manifest names, the
// licence, and a trimmed manifest as a JS module (mods cannot read a JSON file synchronously).
// Usage: node scripts/vendor-icons.mjs <extracted npm package folder>
import { copyFileSync, existsSync, mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { basename, join } from "node:path";
import { fileURLToPath } from "node:url";

const MAPS = ["fileNames", "fileExtensions", "folderNames", "folderNamesExpanded"];

export function trimManifest(manifest, version) {
  const icon = (id) => {
    const def = manifest.iconDefinitions[id];
    if (!def) throw new Error(`manifest names an undefined icon: ${id}`);
    return basename(def.iconPath, ".svg");
  };
  const out = { version, file: icon(manifest.file), folder: icon(manifest.folder), folderExpanded: icon(manifest.folderExpanded) };
  for (const map of MAPS) {
    out[map] = {};
    for (const [key, id] of Object.entries(manifest[map] ?? {})) {
      const lower = key.toLowerCase();
      if (!Object.hasOwn(out[map], lower)) out[map][lower] = icon(id);
    }
  }
  return out;
}

const iconsOf = (t) => [...new Set([t.file, t.folder, t.folderExpanded, ...MAPS.flatMap((m) => Object.values(t[m]))])].sort();

export function vendor({ pkg, mod }) {
  const icons = join(mod, "icons", "material");
  const manifestFile = join(mod, "material-icons.js");
  for (const p of [icons, manifestFile]) if (existsSync(p)) throw new Error(`${p} already exists; move it away first`);
  const { version } = JSON.parse(readFileSync(join(pkg, "package.json"), "utf8"));
  const trimmed = trimManifest(JSON.parse(readFileSync(join(pkg, "dist", "material-icons.json"), "utf8")), version);
  mkdirSync(icons, { recursive: true });
  const names = iconsOf(trimmed);
  for (const name of names) copyFileSync(join(pkg, "icons", `${name}.svg`), join(icons, `${name}.svg`));
  copyFileSync(join(pkg, "LICENSE"), join(icons, "LICENSE"));
  writeFileSync(manifestFile, `// Generated by scripts/vendor-icons.mjs from material-icon-theme ${version} (MIT). Do not edit.\n`
    + `export default ${JSON.stringify(trimmed)};\n`);
  return { version, icons: names.length };
}

if (process.argv[1] === fileURLToPath(import.meta.url)) {
  const pkg = process.argv[2];
  if (!pkg) {
    console.error("usage: node scripts/vendor-icons.mjs <extracted npm package folder>");
    process.exit(2);
  }
  console.log(vendor({ pkg, mod: fileURLToPath(new URL("../mods/latveria-files/", import.meta.url)) }));
}
```

```js
// Picks a Material Icon Theme icon the way VS Code does: the full file name first, then the
// longest extension ("test.mjs" before "mjs"), then the default. Names compare lower-cased.
const DIR = "mods/latveria-files/icons/material";
const path = (icon) => `${DIR}/${icon}.svg`;
const own = (map, key) => (Object.hasOwn(map, key) ? map[key] : undefined);

export function resolveFileIcon(name, manifest) {
  const lower = name.toLowerCase();
  const byName = own(manifest.fileNames, lower);
  if (byName) return path(byName);
  for (let i = lower.indexOf("."); i !== -1; i = lower.indexOf(".", i + 1)) {
    const byExtension = own(manifest.fileExtensions, lower.slice(i + 1));
    if (byExtension) return path(byExtension);
  }
  return path(manifest.file);
}

export function resolveFolderIcon(name, open, manifest) {
  const icon = own(open ? manifest.folderNamesExpanded : manifest.folderNames, name.toLowerCase());
  return path(icon ?? (open ? manifest.folderExpanded : manifest.folder));
}
```
