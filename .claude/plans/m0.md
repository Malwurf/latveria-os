```js
// Fails when the repo carries personal data: names of accounts, clients or people (kept here only as
// sha256, so the scanner itself never brings them in), machine paths and e-mail addresses.
import { createHash } from "node:crypto";
import { execFileSync } from "node:child_process";
import { readFileSync } from "node:fs";
import { fileURLToPath } from "node:url";

export const FORBIDDEN = new Set([
  "ec4a2ae9d8ebb3c1f174f90de8989f22921df51819c9558fec8e74955808aff1",
  "57fd3e98b5fa1345f8474c6277f0b2dfeef52c1d398cdd7e7a661a90195e0020",
  "c8ed1dcd43058307f95a0246d2c63f23f46b1092329117caa8b3090bd27e5d56",
  "a4c5724011b930f43d3ab9cd2ba8d22da81cde2a69ed975ac1855573335e8aeb",
  "66f471705371b5ce331603e4ea14f3fa77a74b45400b9cde46522e7f08dee15d",
  "f6b11133f7e43b27f293ff1da3cc59141e51a276fbbeda0369f22bec74fe61a6",
  "559ac631b69795cda36bfccd088bb9bc7d19f33bb7fa3dadf6b52e3b437b8a9f",
  "5210528c23aedfaed8165119d047bdd43c4c875fb7d9f015fa3a3c07f9452be8", // inflected stem
]);
const MIN = 5;
const MAX = 10;
const PATH = /\/(?:Users\/[A-Za-z]|home\/(?!test\b)[a-z])/;
const EMAIL = /[A-Za-z0-9._%+-]+@([A-Za-z0-9.-]+\.[A-Za-z]{2,})/g;
const ALLOWED_DOMAINS = /(^|\.)(example\.(com|org)|users\.noreply\.github\.com)$/i;
const sha = (s) => createHash("sha256").update(s).digest("hex");

function hasTerm(line, terms) {
  const folded = line.normalize("NFD").replace(/\p{Diacritic}/gu, "").toLowerCase().replace(/ł/g, "l"); // ł has no NFD form
  for (const token of folded.split(/[^a-z0-9]+/)) {
    for (let len = MIN; len <= Math.min(MAX, token.length); len++) {
      for (let i = 0; i + len <= token.length; i++) if (terms.has(sha(token.slice(i, i + len)))) return true;
    }
  }
  return false;
}

export function scanText(path, text, terms = FORBIDDEN) {
  const findings = [];
  text.split("\n").forEach((line, i) => {
    const at = { path, line: i + 1 };
    if (hasTerm(line, terms)) findings.push({ ...at, kind: "term" });
    if (PATH.test(line)) findings.push({ ...at, kind: "path" });
    for (const m of line.matchAll(EMAIL)) if (!ALLOWED_DOMAINS.test(m[1])) { findings.push({ ...at, kind: "email" }); break; }
  });
  return findings;
}

function repoFiles() {
  const out = execFileSync("git", ["ls-files", "--cached", "--others", "--exclude-standard", "-z"], { encoding: "utf8" });
  return out.split("\0").filter(Boolean);
}

if (process.argv[1] === fileURLToPath(import.meta.url)) {
  const findings = [];
  for (const path of repoFiles()) {
    let buf;
    try { buf = readFileSync(path); } catch { continue; } // deleted but still listed
    if (buf.subarray(0, 8000).includes(0)) continue; // binary
    findings.push(...scanText(path, buf.toString("utf8")));
  }
  for (const f of findings) console.error(`${f.path}:${f.line}: ${f.kind}`);
  console.log(findings.length ? `privacy: ${findings.length} finding(s)` : "privacy: clean");
  process.exit(findings.length ? 1 : 0);
}
```

```js
import { test } from "node:test";
import assert from "node:assert/strict";
import { execFileSync } from "node:child_process";
import { readFileSync } from "node:fs";

// Rule 1: no credentials, no network. Rule 2: read-only towards repositories, nothing deleted.
const FORBIDDEN = [
  [/\bfetch\s*\(/, "network call"],
  [/api\.anthropic|api\.openai/i, "AI provider API"],
  [/oauth/i, "OAuth"],
  [/find-(generic|internet)-password|["'`]security["'`]/, "Keychain"],
  [/\b(unlink|unlinkSync|rmSync|rmdirSync)\s*\(|\brm\s+-/, "deletes files"],
  [/\bgit\s+(push|merge|commit|rebase|reset|clean)\b/, "writes to a repository"],
];
const EXEC = /\bexec(?:File)?(?:Sync)?\(\s*["'`]([\w-]+)["'`]\s*(?:,\s*\[\s*["'`]([\w-]+)["'`])?/g;
const ALLOWED_PROGRAMS = new Set(["git", "sh", "hc"]);
const ALLOWED_GIT = new Set(["status", "rev-parse", "ls-files", "log", "diff", "show"]);

export function violations(text) {
  const found = [];
  for (const [re, why] of FORBIDDEN) if (re.test(text)) found.push(why);
  for (const [, program, sub] of text.matchAll(EXEC)) {
    if (!ALLOWED_PROGRAMS.has(program)) found.push(`runs ${program}`);
    else if (program === "git" && !ALLOWED_GIT.has(sub)) found.push(`git ${sub ?? "?"}`);
  }
  return found;
}

test("catches each forbidden kind of call", () => {
  assert.deepEqual(violations('await fetch("https://x")'), ["network call"]);
  assert.deepEqual(violations('exec("security", ["find-generic-password"])'), ["Keychain", "runs security"]);
  assert.deepEqual(violations('exec("git", ["push", "origin"])'), ["git push"]);
  assert.deepEqual(violations('exec("curl", ["x"])'), ["runs curl"]);
  assert.deepEqual(violations("fs.rmSync(dir)"), ["deletes files"]);
  assert.deepEqual(violations('exec("sh", ["-c", "git push origin main"])'), ["writes to a repository"]);
});

test("allows read-only git and hc", () => {
  assert.deepEqual(violations('exec("git", ["status", "--porcelain"]); exec("hc", ["label"])'), []);
  assert.deepEqual(violations('execFileSync("git", ["ls-files", "-z"])'), []);
});

test("no code in the repo breaks rules 1 and 2", () => {
  const files = execFileSync("git", ["ls-files", "--cached", "--others", "--exclude-standard", "-z"], { encoding: "utf8" })
    .split("\0").filter((f) => /\.(m?js)$/.test(f) && !f.startsWith("test/") && f !== "eslint.config.mjs");
  const broken = files.flatMap((f) => violations(readFileSync(f, "utf8")).map((why) => `${f}: ${why}`));
  assert.deepEqual(broken, []);
});
```

**Code:** `lib/contrast.mjs`
```js
// WCAG 2.x relative luminance and contrast ratio, for #rrggbb colors.
const HEX = /^#[0-9a-f]{6}$/i;
const channels = (hex) => {
  if (!HEX.test(hex)) throw new TypeError(`expected #rrggbb, got ${hex}`);
  return [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16));
};
const linear = (c) => (c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4);
const luminance = (hex) => {
  const [r, g, b] = channels(hex).map((c) => linear(c / 255));
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
};

export function contrastRatio(a, b) {
  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return (hi + 0.05) / (lo + 0.05);
}

// `fg` drawn at `alpha` over `bg`, as GPUI draws an "#rrggbbaa" fill.
export function mix(fg, bg, alpha) {
  const [f, b] = [channels(fg), channels(bg)];
  return "#" + f.map((c, i) => Math.round(c * alpha + b[i] * (1 - alpha)).toString(16).padStart(2, "0")).join("");
}
```

```js
import { test } from "node:test";
import assert from "node:assert/strict";
import { readFileSync, readdirSync } from "node:fs";
import { contrastRatio, mix } from "../lib/contrast.mjs";

const TOKENS = ["canvas", "chrome", "surface", "popover", "raised", "hover", "selected", "border", "input", "muted", "dim",
  "accent", "accent_foreground", "solid", "solid_foreground", "working", "blocked", "done", "idle", "unknown"];
const TEXT = ["foreground", "muted", "dim", "accent", "working", "blocked", "done", "idle", "unknown"];
const GROUNDS = ["background", "canvas", "chrome", "surface", "popover", "raised", "hover", "selected"];
const STATUS = ["working", "blocked", "done", "idle"];
const TINT = 0x26 / 255; // mods draw status badges as `${color}26`
const MIN = 4.5;

const themes = readdirSync(new URL("../themes/", import.meta.url)).filter((f) => f.endsWith(".json"))
  .map((f) => [f, JSON.parse(readFileSync(new URL(`../themes/${f}`, import.meta.url), "utf8"))]);
const flat = (t) => ({ background: t.background, foreground: t.foreground, selection: t.selection, ...t.colors });

export function unreadablePairs(t) {
  const c = flat(t);
  const pairs = [];
  for (const fg of TEXT) for (const bg of GROUNDS) pairs.push([`${fg}/${bg}`, c[fg], c[bg]]);
  for (const s of STATUS) pairs.push([`${s}/tint@chrome`, c[s], mix(c[s], c.chrome, TINT)]);
  pairs.push(["accent_foreground/accent", c.accent_foreground, c.accent], ["solid_foreground/solid", c.solid_foreground, c.solid],
    ["foreground/selection", c.foreground, c.selection]);
  t.ansi.forEach((a, i) => { if (i > 0) pairs.push([`ansi${i}/background`, a, c.background]); });
  return pairs.filter(([, f, b]) => contrastRatio(f, b) < MIN).map(([name, f, b]) => `${name} ${contrastRatio(f, b).toFixed(2)}`);
}

test("there is a latveria theme", () => assert.ok(themes.some(([f]) => f === "latveria.json")));

for (const [file, t] of themes) {
  test(`${file}: defines every color token as #rrggbb`, () => {
    for (const key of ["background", "foreground", "selection", "cursor"]) assert.match(t[key], /^#[0-9a-f]{6}$/i, key);
    assert.equal(t.ansi.length, 16);
    for (const a of t.ansi) assert.match(a, /^#[0-9a-f]{6}$/i);
    for (const k of TOKENS) assert.match(t.colors?.[k] ?? "", /^#[0-9a-f]{6}$/i, `colors.${k}`);
```

**Code:** `install.mjs`
```js
#!/usr/bin/env node
// Copies the theme and layout into Herdr's home. Never overwrites silently: a file the person changed
// since our last install is kept as <name>.bak-<ms> first. Nothing is deleted.
import { createHash } from "node:crypto";
import { copyFileSync, existsSync, mkdirSync, readFileSync, readdirSync, writeFileSync } from "node:fs";
import { homedir } from "node:os";
import { dirname, join } from "node:path";
import { fileURLToPath } from "node:url";

const PARTS = ["themes", "layouts"];
const sha = (path) => createHash("sha256").update(readFileSync(path)).digest("hex");

export function install({ repo, herdrHome, stateDir, dryRun = false, now = Date.now() }) {
  const ledgerPath = join(stateDir, "installed.json");
  const ledger = existsSync(ledgerPath) ? JSON.parse(readFileSync(ledgerPath, "utf8")) : {};
  const steps = [];
  for (const part of PARTS) {
    const from = join(repo, part);
    if (!existsSync(from)) continue;
    for (const name of readdirSync(from).filter((n) => n.endsWith(".json"))) {
      const rel = `${part}/${name}`;
      const src = join(from, name);
      const dest = join(herdrHome, part, name);
      const hash = sha(src);
      let action = "create";
      if (existsSync(dest)) {
        const current = sha(dest);
        action = current === hash ? "same" : ledger[rel] === current ? "update" : "backup";
      }
      steps.push({ rel, action });
      if (dryRun) continue;
      if (action !== "same") {
        mkdirSync(dirname(dest), { recursive: true });
        if (action === "backup") copyFileSync(dest, `${dest}.bak-${now}`);
        copyFileSync(src, dest);
      }
      ledger[rel] = hash;
    }
  }
  if (!dryRun) {
    mkdirSync(stateDir, { recursive: true });
    writeFileSync(ledgerPath, JSON.stringify(ledger, null, 2) + "\n");
  }
  return steps;
}

if (process.argv[1] === fileURLToPath(import.meta.url)) {
  const steps = install({
    repo: dirname(fileURLToPath(import.meta.url)),
    herdrHome: process.env.HERDR_APP_HOME ?? join(homedir(), ".config", "herdr-app"),
    stateDir: process.env.LATVERIA_STATE ?? join(homedir(), ".config", "latveria"),
    dryRun: process.argv.includes("--dry-run"),
  });
  for (const s of steps) console.log(`${s.action.padEnd(6)} ${s.rel}`);
}
```
